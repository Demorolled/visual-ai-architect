using System.Collections.Generic;
using System.IO;

public static class StorageOperations {
    public static void SaveTodos(List<string> todos, string filePath) {
        File.WriteAllLines(filePath, todos);
    }

    public static List<string> LoadTodos(string filePath) {
        if (!File.Exists(filePath)) return new List<string>();
        return File.ReadAllLines(filePath).ToList();
    }
}
