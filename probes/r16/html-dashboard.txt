<script>
const weatherData = {
  'London': { temp: [16, 13, 19, 22, 26, 24, 21], icon: '☁️', minTemp: 13, maxTemp: 26 },
  'Paris': { temp: [18, 14, 20, 24, 28, 26, 22], icon: '☀️', minTemp: 14, maxTemp: 28 },
  'Tokyo': { temp: [22, 18, 25, 28, 32, 30, 24], icon: '☀️', minTemp: 18, maxTemp: 32 },
  'New York': { temp: [13, 9, 17, 21, 25, 22, 18], icon: '❄️', minTemp: 9, maxTemp: 25 },
  'Sydney': { temp: [26, 23, 30, 33, 37, 35, 31], icon: '☀️', minTemp: 23, maxTemp: 37 },
  'Berlin': { temp: [12, 9, 16, 20, 24, 22, 18], icon: '❄️', minTemp: 9, maxTemp: 24 },
  'Singapore': { temp: [29, 26, 32, 35, 39, 37, 33], icon: '☀️', minTemp: 26, maxTemp: 39 },
};

function drawChart(ctx, data) {
  const maxTemp = Math.max(...data.temp);
  const minTemp = Math.min(...data.temp);
  const chartHeight = 250;

  // Clear canvas
  ctx.clearRect(0, 0, ctx.canvas.width, ctx.canvas.height);

  // Draw background grid
  for (let i = 0; i <= 7; i++) {
    ctx.strokeStyle = `rgba(255,255,255,${(0.05 + i * 0.15).toFixed(2)}`;
    ctx.lineWidth = 0.5;
    if (i === 6) ctx.lineWidth = 1.5;
    ctx.beginPath();
    ctx.moveTo(i * 35, chartHeight);
    ctx.lineTo(i * 35, chartHeight - (maxTemp - data.temp[i]) * 10);
    ctx.stroke();
    if (i === 6) { ctx.font = '10px Arial'; ctx.fillStyle = '#666'; ctx.fillText('Today', i * 35, chartHeight + 8); }
  }

  // Draw temperature lines
  for (let t = minTemp; t <= maxTemp; t += 5) {
    ctx.strokeStyle = `rgba(255,255,255,${(0.05 + (maxTemp - t) * 0.03).toFixed(2)}`;
    ctx.lineWidth = 0.5;
    ctx.beginPath();
    ctx.moveTo(0, chartHeight - t * 10);
    ctx.lineTo(245, chartHeight - t * 10);
    ctx.stroke();
    if (t === 20) { ctx.font = '8px Arial'; ctx.fillStyle = '#666'; ctx.fillText(t, 0, chartHeight - t * 10 - 4); }
  }

  // Draw data points
  data.temp.forEach((temp, i) => {
    const x = i * 35;
    const y = chartHeight - temp * 10;
    ctx.fillStyle = '#00d9ff';
    ctx.beginPath();
    ctx.arc(x, y, 5, 0, Math.PI * 2);
    ctx.fill();
  });

  // Draw lines
  data.temp.forEach((temp, i) => {
    const x = i * 35;
    const y = chartHeight - temp * 10;
    ctx.strokeStyle = '#00d9ff';
    ctx.lineWidth = 2;
    ctx.beginPath();
    ctx.moveTo(x, y);
    ctx.lineTo(x + 20, y);
    ctx.stroke();
  });
}

function fetchWeather() {
  const city = document.getElementById('searchInput').value.trim().toLowerCase();
  if (!city) return;
  if (weatherData[city]) {
    const card = document.querySelector('.card');
    const icon = document.querySelector('.weather-icon');
    const tempDisplay = document.querySelector('.temp');
    const feelsLike = document.querySelector('.feels-like');
    const forecast = document.querySelectorAll('.forecast-card');
    
    card.querySelector('h2').textContent = city;
    icon.textContent = weatherData[city].icon;
    tempDisplay.textContent = `${weatherData[city].temp[0]}°C`;
    feelsLike.textContent = `Feels like ${weatherData[city].temp[0]}°C`;
    const forecastData = weatherData[city].temp.slice(1, 6);
    forecast.forEach((el, i) => {
      el.querySelector('.day').textContent = [0, 'Monday', 'Tuesday', 'Wednesday', 'Thursday', 'Friday', 'Saturday'][i];
      el.querySelector('.temp').textContent = `${forecastData[i]}°C`;
      el.querySelector('.high-low').textContent = `${forecastData[i]}°C / ${forecastData[i] + 3}°C`;
    });
    const chartCtx = document.getElementById('chartCanvas').getContext('2d');
    drawChart(chartCtx, weatherData[city]);
  } else {
    alert('City not found');
  }
}
